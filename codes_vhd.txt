library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.STD_LOGIC_ARITH.ALL;
use IEEE.STD_LOGIC_UNSIGNED.ALL;

entity alu_4bit is
    port (
        A   : in  STD_LOGIC_VECTOR(3 downto 0);
        B   : in  STD_LOGIC_VECTOR(3 downto 0);
        SEL : in  STD_LOGIC_VECTOR(2 downto 0);
        Y   : out STD_LOGIC_VECTOR(3 downto 0)
    );
end alu_4bit;

architecture Behavioral of alu_4bit is
begin
    process (A, B, SEL)
    begin
        case SEL is
            when "000"  => Y <= A + B;
            when "001"  => Y <= A - B;
            when "010"  => Y <= A or B;
            when "011"  => Y <= A xor B;
            when "100"  => Y <= A and B;
            when "101"  => Y <= A xnor B;
            when "110"  => Y <= A nand B;
            when "111"  => Y <= not A;
            when others => Y <= "0000";
        end case;
    end process;
end Behavioral;

use IEEE.STD_LOGIC_UNSIGNED.ALL;

entity tb_alu_4bit is
end tb_alu_4bit;

architecture Behavioral of tb_alu_4bit is

    component alu_4bit
        port (
            A   : in  STD_LOGIC_VECTOR(3 downto 0);
            B   : in  STD_LOGIC_VECTOR(3 downto 0);
            SEL : in  STD_LOGIC_VECTOR(2 downto 0);
            Y   : out STD_LOGIC_VECTOR(3 downto 0)
        );
    end component;

    signal A   : STD_LOGIC_VECTOR(3 downto 0) := "0000";
    signal B   : STD_LOGIC_VECTOR(3 downto 0) := "0000";
    signal SEL : STD_LOGIC_VECTOR(2 downto 0) := "000";
    signal Y   : STD_LOGIC_VECTOR(3 downto 0);

begin

    uut: alu_4bit port map (A => A, B => B, SEL => SEL, Y => Y);

    stim_proc: process
    begin
        -- A = 9, B = 5, test all 8 operations
        A <= "1001"; B <= "0101";

        SEL <= "000"; wait for 10 ns;   -- A + B     = 14 (1110)
        SEL <= "001"; wait for 10 ns;   -- A - B     = 4  (0100)
        SEL <= "010"; wait for 10 ns;   -- A or B    = 13 (1101)
        SEL <= "011"; wait for 10 ns;   -- A xor B   = 12 (1100)
        SEL <= "100"; wait for 10 ns;   -- A and B   = 1  (0001)
        SEL <= "101"; wait for 10 ns;   -- A xnor B  = 3  (0011)
        SEL <= "110"; wait for 10 ns;   -- A nand B  = 14 (1110)
        SEL <= "111"; wait for 10 ns;   -- not A     = 6  (0110)

        -- Overflow test: 15 + 1 wraps to 0
        A <= "1111"; B <= "0001";
        SEL <= "000"; wait for 10 ns;   -- result 0 (0000)

        -- Zero result: 3 - 3
        A <= "0011"; B <= "0011";
        SEL <= "001"; wait for 10 ns;   -- result 0 (0000)

        wait;   -- stop
    end process;

end Behavioral;

IEEE.NUMERIC_STD.ALL;

N : integer := 4
);
port (

clk : in STD_LOGIC; reset : in
STD_LOGIC;
count : out STD_LOGIC_VECTOR(1 downto 0)
);
end mod_n_counter;
architecture Behavioral of mod_n_counter is signal count_int :
integer range 0 to N-1 := 0; begin
process(clk) begin
if rising_edge(clk) then if reset =
'1' then count_int <= 0;
elsif count_int = N-1 then count_int
<= 0;

else

end if;
end if;
count_int <= count_int + 1;

end process;
count <= std_logic_vector(to_unsigned(count_int, count'length)); end
Behavioral;

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.STD_LOGIC_ARITH.ALL;
use IEEE.STD_LOGIC_UNSIGNED.ALL;

entity shift_reg_4bit is
    port (
        clk     : in  STD_LOGIC;
        reset   : in  STD_LOGIC;                      -- direct overriding clear
        datain  : in  STD_LOGIC_VECTOR(3 downto 0);   -- parallel input
        SL_in   : in  STD_LOGIC;                      -- serial input for left shift
        SR_in   : in  STD_LOGIC;                      -- serial input for right shift
        sel     : in  STD_LOGIC_VECTOR(1 downto 0);   -- sel(1)=S1, sel(0)=S0
        dataout : out STD_LOGIC_VECTOR(3 downto 0)    -- parallel output (QA..QD)
    );
end shift_reg_4bit;

architecture Behavioral of shift_reg_4bit is
    signal q : STD_LOGIC_VECTOR(3 downto 0) := "0000";
begin
    process (clk, reset)
    begin
        if reset = '1' then
            q <= "0000";                              -- asynchronous clear
        elsif rising_edge(clk) then
            case sel is
                when "00"   => q <= q;                          -- hold (inhibit clock)
                when "01"   => q <= SR_in & q(3 downto 1);      -- shift right
                when "10"   => q <= q(2 downto 0) & SL_in;      -- shift left
                when "11"   => q <= datain;                     -- parallel load
                when others => q <= q;
            end case;
        end if;
    end process;

    dataout <= q;
end Behavioral;

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity tb_shift_reg_4bit is
end tb_shift_reg_4bit;

architecture Behavioral of tb_shift_reg_4bit is

    component shift_reg_4bit
        port (
            clk     : in  STD_LOGIC;
            reset   : in  STD_LOGIC;
            datain  : in  STD_LOGIC_VECTOR(3 downto 0);
            SL_in   : in  STD_LOGIC;
            SR_in   : in  STD_LOGIC;
            sel     : in  STD_LOGIC_VECTOR(1 downto 0);
            dataout : out STD_LOGIC_VECTOR(3 downto 0)
        );
    end component;

    signal clk     : STD_LOGIC := '0';
    signal reset   : STD_LOGIC := '0';
    signal datain  : STD_LOGIC_VECTOR(3 downto 0) := "0000";
    signal SL_in   : STD_LOGIC := '0';
    signal SR_in   : STD_LOGIC := '0';
    signal sel     : STD_LOGIC_VECTOR(1 downto 0) := "00";
    signal dataout : STD_LOGIC_VECTOR(3 downto 0);

    constant clk_period : time := 10 ns;

begin

    -- Unit Under Test
    uut: shift_reg_4bit
        port map (
            clk     => clk,
            reset   => reset,
            datain  => datain,
            SL_in   => SL_in,
            SR_in   => SR_in,
            sel     => sel,
            dataout => dataout
        );

    -- Clock generation
    clk_process : process
    begin
        clk <= '0';
        wait for clk_period/2;
        clk <= '1';
        wait for clk_period/2;
    end process;

    -- Stimulus
    stim_proc : process
    begin
        -- 1. Asynchronous reset
        reset <= '1';
        wait for 22 ns;
        reset <= '0';
        wait until falling_edge(clk);
        assert dataout = "0000" report "Reset failed" severity error;

        -- 2. Parallel load (S1S0 = 11), load 1010
        sel    <= "11";
        datain <= "1010";
        wait until falling_edge(clk);
        assert dataout = "1010" report "Parallel load failed" severity error;

        -- 3. Hold / inhibit clock (S1S0 = 00)
        sel <= "00";
        wait until falling_edge(clk);
        assert dataout = "1010" report "Hold failed" severity error;

        -- 4. Shift right (S1S0 = 01), SR_in = 1, two clocks
        sel   <= "01";
        SR_in <= '1';
        wait until falling_edge(clk);
        assert dataout = "1101" report "Shift right 1 failed" severity error;
        wait until falling_edge(clk);
        assert dataout = "1110" report "Shift right 2 failed" severity error;

        --    Shift right, SR_in = 0, two clocks
        SR_in <= '0';
        wait until falling_edge(clk);
        assert dataout = "0111" report "Shift right 3 failed" severity error;
        wait until falling_edge(clk);
        assert dataout = "0011" report "Shift right 4 failed" severity error;

        -- 5. Shift left (S1S0 = 10), SL_in = 1, two clocks
        sel   <= "10";
        SL_in <= '1';
        wait until falling_edge(clk);
        assert dataout = "0111" report "Shift left 1 failed" severity error;
        wait until falling_edge(clk);
        assert dataout = "1111" report "Shift left 2 failed" severity error;

        --    Shift left, SL_in = 0, two clocks
        SL_in <= '0';
        wait until falling_edge(clk);
        assert dataout = "1110" report "Shift left 3 failed" severity error;
        wait until falling_edge(clk);
        assert dataout = "1100" report "Shift left 4 failed" severity error;

        -- 6. Load again, then assert reset in the middle of a clock cycle
        sel    <= "11";
        datain <= "0101";
        wait until falling_edge(clk);
        assert dataout = "0101" report "Second load failed" severity error;

        wait for 2 ns;
        reset <= '1';                 -- async clear, no clock edge needed
        wait for 1 ns;
        assert dataout = "0000" report "Async reset failed" severity error;
        wait for 10 ns;
        reset <= '0';

        wait for 20 ns;
        report "Simulation finished" severity note;
        wait;
    end process;

end Behavioral;

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.STD_LOGIC_ARITH.ALL;
use IEEE.STD_LOGIC_UNSIGNED.ALL;
entity fifo is
port ( clk : in std_logic;
write_addr : in std_logic_vector(3 downto 0);
enr : in std_logic; --enable read,should be '0' when not in use.
enw : in std_logic; --enable write,should be '0' when not in use.
dataout : out std_logic_vector(7 downto 0); --output data
datain : in std_logic_vector (7 downto 0); --input data
empty : out std_logic; --set as '1' when the queue is empty
full : out std_logic --set as '1' when the queue is full
);
end fifo;
architecture Behavioral of fifo is
type memory_type is array (0 to 63) of std_logic_vector(7 downto 0);
signal memory : memory_type :=(others => (others => '0')); --memory for queue.
signal readptr,writeptr : std_logic_vector(3 downto 0) :="0000"; --read/write pointers.
signal count : std_logic_vector(23 downto 0);
signal sclk : std_logic;
begin
writeptr<= write_addr;
process(clk)
begin
if(clk'event and clk='1') then
count<= count + 1;
end if;
end process;
sclk<= count(23);
process(sclk)
begin
if(sclk'event and sclk='1' and enw ='1') then
memory(conv_integer(writeptr)) <= datain;
end if;
if(sclk'event and sclk='1' and enr ='1') then
dataout<= memory(conv_integer(readptr));
readptr<= readptr + '1'; --points to next address.
end if;
if(readptr = "1111") then --resetting read pointer.
readptr<= "0000";

end if;
if(writeptr = "1111") then --checking whether queue is full or not
full<='1';
else
full<='0';
end if;
if(writeptr = "0000") then --checking whether queue is empty or not
empty<='1';
else
empty<='0';
end if;
end process;
end Behavioral;
